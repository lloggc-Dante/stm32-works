/*==============================================================================
 * 3-2 LED流水灯 —— 逐行注释版（参考用，不要加进 Keil 工程）
 * 对应文件：User\main.c（当前是 8 灯轮流亮的流水灯版本）
 * 芯片：STM32F103C8T6，主频 72MHz
 *
 * 【说明】本文件只是给你看的文本，不是源码。Keil 只编译工程 Groups 里列出的
 *  .c 文件，这个 .txt 不会被编译，也不会和 main.c 冲突。
 *  main.c 本身一个字都没改。
 *
 * 【和 3-1「LED闪烁」的区别，只有两点】
 *   ① 引脚范围：3-1 配 1 个引脚（GPIO_Pin_0），这里配全部 16 个（GPIO_Pin_All）
 *   ② 输出方式：3-1 用 GPIO_WriteBit 一次改一个引脚，
 *                这里用 GPIO_Write 一次写整个端口 16 位 → "流水"就是让那个 0 依次挪位置
 *   其余（开时钟、填配置单、GPIO_Init）完全一样。
 *============================================================================*/

#include "stm32f10x.h"                  // Device header（同 3-1：寄存器地址 + 库函数声明 + 各种宏）
#include "Delay.h"                      // 自己写的延时模块（System\Delay.c 里实现）

int main(void)
{
	/*--------------------------------------------------------------------------
	 * 第①步：打开 GPIOA 的时钟（和 3-1 完全一样）
	 *--------------------------------------------------------------------------
	 * RCC_APB2PeriphClockCmd(要开哪个外设, ENABLE/DISABLE);
	 * GPIOA 挂 APB2 总线 → 用 RCC_APB2Periph_GPIOA。
	 * 复位后外设时钟默认关闭，不开时钟后面写寄存器全部无效（且不报错）。
	 * 详见 3-1 的 main_注释版.txt。
	 *------------------------------------------------------------------------*/
	RCC_APB2PeriphClockCmd(RCC_APB2Periph_GPIOA, ENABLE);

	/*--------------------------------------------------------------------------
	 * 第②步：填配置单（结构体）
	 *--------------------------------------------------------------------------
	 * GPIO_InitTypeDef 三个成员：GPIO_Pin / GPIO_Speed / GPIO_Mode。
	 * 这一步只是在内存里准备数据，不碰硬件。
	 *------------------------------------------------------------------------*/
	GPIO_InitTypeDef GPIO_InitStructure;

	GPIO_InitStructure.GPIO_Mode = GPIO_Mode_Out_PP;      // 推挽输出：能主动输出高 3.3V / 低 0V
	GPIO_InitStructure.GPIO_Speed = GPIO_Speed_50MHz;     // 输出带宽档位（10/2/50MHz），点灯随意

	/*--------------------------------------------------------------------------
	 * ★ 与 3-1 的第一个区别：GPIO_Pin_All
	 *--------------------------------------------------------------------------
	 *   GPIO_Pin_0  = 0x0001  → 只选 PA0
	 *   GPIO_Pin_All = 0xFFFF → 一次选中 PA0 ~ PA15 全部 16 个引脚
	 *   （定义在 Library\stm32f10x_gpio.h 第 143 行：((uint16_t)0xFFFF)）
	 *
	 * 所以这一句执行后，GPIOA 的 16 个引脚**全部**被设成推挽输出。
	 * 这样后面无论往哪一位写 0，那个引脚都能输出低电平。
	 *
	 * ⚠️ 提醒（以后会踩到的坑）：PA13 / PA14 是 ST-Link 的 SWDIO / SWCLK，
	 *   这里也被一起配成了推挽输出，程序跑起来后调试口就被占用了。
	 *   如果哪天下载或在线仿真报"连不上目标 / Cannot access target"，
	 *   解决办法：Options for Target → Debug → Settings → Connect 选
	 *   "under Reset"（复位时连接），或者按住板上复位键再点下载。
	 *------------------------------------------------------------------------*/
	GPIO_InitStructure.GPIO_Pin = GPIO_Pin_All;

	/*--------------------------------------------------------------------------
	 * 第③步：把配置单写进寄存器
	 *--------------------------------------------------------------------------
	 * GPIO_Init(GPIOA, &GPIO_InitStructure);
	 *   参数1 = 哪组端口；参数2 = 结构体地址（& 取地址）。
	 *   内部按内容改 CRL / CRH 寄存器，16 个引脚全部生效（详见 3-1 注释）。
	 *------------------------------------------------------------------------*/
	GPIO_Init(GPIOA, &GPIO_InitStructure);

	/*--------------------------------------------------------------------------
	 * 第④步：主循环 —— 让"低电平"在 8 个灯之间依次移动，形成流水灯
	 *------------------------------------------------------------------------*/
	while(1)
	{
		/*----------------------------------------------------------------------
		 * ★ 与 3-1 的第二个区别：GPIO_Write —— 一次写整个端口的 16 位
		 *----------------------------------------------------------------------
		 * 函数原型（Library\stm32f10x_gpio.h 第 360 行）：
		 *     void GPIO_Write(GPIO_TypeDef* GPIOx, uint16_t PortVal);
		 *
		 *   参数1 GPIOA  ：操作哪组端口（这里是整组一起写！）
		 *   参数2 PortVal：16 位数据，每 1 位对应 1 个引脚
		 *                  bit0 = PA0，bit1 = PA1 …… bit15 = PA15
		 *                  **写 1 → 该引脚输出高电平；写 0 → 输出低电平**
		 *
		 *   内部实现就一句（stm32f10x_gpio.c 第 422 行）：
		 *         GPIOx->ODR = PortVal;
		 *   ODR = Output Data Register，输出数据寄存器，它保存着 16 个引脚
		 *   当前该输出什么电平。整寄存器一次性覆盖写。
		 *
		 * ★ ~0x0001 是什么意思（本行重点）
		 *   0x0001 的二进制：0000 0000 0000 0001   （只有 bit0 = 1）
		 *   ~ 是按位取反   ：1111 1111 1111 1110   （只有 bit0 = 0）
		 *   所以 PA0 = 0 → 低电平 → PA0 上的 LED 亮
		 *        其余 15 个引脚 = 1 → 高电平 → 其余 LED 全灭
		 *   由于 LED 是低电平点亮，"哪一位是 0，哪个灯就亮"，
		 *   于是只要让这个 0 依次挪到 bit0~bit7，就是 8 个灯轮流亮的流水灯。
		 *
		 *   小知识：0x0001 本身是 int（32 位），~0x0001 = 0xFFFFFFFE，
		 *   传给 uint16_t 类型的参数时自动截断成 0xFFFE，效果一样，不必担心。
		 *
		 * ★ 对比记忆（三种写引脚的方式，用途不同）：
		 *   GPIO_Write(GPIOA, 0xFFFE);                        // 整端口 16 位一起写，其它位被覆盖
		 *   GPIO_WriteBit(GPIOA, GPIO_Pin_0, Bit_RESET);      // 只改 PA0，其它引脚不动（3-1 用的）
		 *   GPIO_SetBits / GPIO_ResetBits(GPIOA, GPIO_Pin_0); // 只把指定位置 1 / 清 0，其它不动
		 *   ── 想让"多个灯同时按不同状态亮"，就用 GPIO_Write 一次写好整张表；
		 *      想单独点某一个灯，用后两种更安全（不会误改别的引脚）。
		 *--------------------------------------------------------------------*/
		GPIO_Write(GPIOA, ~0x0001);   // 0xFFFE：只有 PA0 为低 → LED1 亮
		Delay_ms(500);                // 保持 500ms

		GPIO_Write(GPIOA, ~0x0002);   // 0xFFFD：只有 PA1 为低 → LED2 亮
		Delay_ms(500);

		GPIO_Write(GPIOA, ~0x0004);   // 0xFFFB：只有 PA2 为低 → LED3 亮
		Delay_ms(500);

		GPIO_Write(GPIOA, ~0x0008);   // 0xFFF7：只有 PA3 为低 → LED4 亮
		Delay_ms(500);

		GPIO_Write(GPIOA, ~0x0010);   // 0xFFEF：只有 PA4 为低 → LED5 亮
		Delay_ms(500);

		GPIO_Write(GPIOA, ~0x0020);   // 0xFFDF：只有 PA5 为低 → LED6 亮
		Delay_ms(500);

		GPIO_Write(GPIOA, ~0x0040);   // 0xFFBF：只有 PA6 为低 → LED7 亮
		Delay_ms(500);

		GPIO_Write(GPIOA, ~0x0080);   // 0xFF7F：只有 PA7 为低 → LED8 亮
		Delay_ms(500);

		/* 执行到这里，while(1) 让程序回到上面第一句，从 LED1 重新开始，
		 * 于是 8 个灯不停地"跑"下去，这就是流水灯。 */
	}
}

/*==============================================================================
 * 附一：上面 8 句其实可以写成循环（等你熟悉 for 循环后可以改成这样，效果一样）
 *----------------------------------------------------------------------------
 *     uint8_t i;
 *     while (1)
 *     {
 *         for (i = 0; i < 8; i++)
 *         {
 *             GPIO_Write(GPIOA, ~(0x0001 << i));   // 0x0001 左移 i 位再取反
 *             Delay_ms(500);
 *         }
 *     }
 *   0x0001 << 0 = 0x0001 → PA0 亮
 *   0x0001 << 1 = 0x0002 → PA1 亮
 *   ……
 *   0x0001 << 7 = 0x0080 → PA7 亮
 *   写 8 句和写 1 个循环，机器执行的效果完全一样，但循环更好改（比如改成 16 个灯）。
 *
 * 附二：本工程用到的库函数速查
 *----------------------------------------------------------------------------
 * RCC_APB2PeriphClockCmd(外设宏, ENABLE/DISABLE)    stm32f10x_rcc.c   开/关 APB2 外设时钟
 * GPIO_Init(端口, &配置结构体)                       stm32f10x_gpio.c  按配置单设置引脚模式
 * GPIO_Write(端口, 16位数据)                         stm32f10x_gpio.c  整端口写 ODR（本次用）
 * GPIO_WriteBit(端口, 引脚掩码, Bit_SET/Bit_RESET)   stm32f10x_gpio.c  只改指定引脚
 * GPIO_SetBits / GPIO_ResetBits(端口, 引脚掩码)      stm32f10x_gpio.c  指定引脚置 1 / 清 0
 * Delay_ms(毫秒) / Delay_us(微秒) / Delay_s(秒)      System\Delay.c    自己写的阻塞式延时
 *
 * 附三：为什么是 8 个灯、接在哪里
 *   你这 8 句写到 ~0x0080（bit7）为止，说明板子上 8 颗 LED 接在 PA0~PA7
 *   （和 3-1 点的那颗 PA0 是同一排），低电平点亮。
 *   如果哪天换成接 PA8~PA15 的板子，就把 ~0x0001 改成 ~0x0100 起，依次左移。
 *
 * 附四：想查函数原型
 *   在 Keil 里把光标放在函数名上按 F12（Go to Definition）跳到定义；
 *   本次每个函数的行号都已经写在上面注释里了。
 *============================================================================*/
